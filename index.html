       const jenjang = document.getElementById('jenjang').value;
       const kelas = document.getElementById('kelas').value;
       const semester = document.getElementById('semester').value;
       const kurikulum = document.getElementById('kurikulum').value;
       const mapel = document.getElementById('mapel').value.trim();
       const materi = document.getElementById('materi').value.trim();
       const waktu = document.getElementById('alokasiWaktu').value.trim();
       const tujuan = document.getElementById('tujuan').value.trim();
       const pemantik = document.getElementById('pemantik').value.trim();
       const model = document.getElementById('model').value;
       const metode = document.getElementById('metode').value.trim();

       }

       const profilChecks = document.querySelectorAll('input[name="profilLulusan"]:checked');
       let profilHTML = '';

           profilChecks.forEach(chk => { profilHTML += '<li>' + chk.value + '</li>'; });
       } else {
           profilHTML = '<li>1. Keimanan & Ketaqwaan, 3. Penalaran Kritis, 5. Kolaborasi & Empati</li>';
       }

       const pancaCintaChecks = document.querySelectorAll('input[name="pancaCinta"]:checked');
       let pancaCintaHTML = '';
       if (pancaCintaChecks.length > 0) {
           pancaCintaChecks.forEach(chk => { pancaCintaHTML += '<li>' + chk.value + '</li>'; });
       } else {
           pancaCintaHTML = '<li>Cinta Allah & Rasul, serta Cinta Sesama Manusia</li>';
       }

       document.getElementById('outKopInstansi').innerText = 'SATUAN PENDIDIKAN ' + jenjang.toUpperCase() + ' - ' + kurikulum.toUpperCase();
       document.getElementById('outHeaderTitle').innerText = 'MODUL AJAR ' + mapel.toUpperCase() + ' - ' + kurikulum.toUpperCase();
       document.getElementById('resGuru').innerText = guru;
       document.getElementById('resSekolah').innerText = sekolah;
       document.getElementById('resJenjangKelas').innerText = jenjang + ' / ' + kelas;
       document.getElementById('resFaseSem').innerText = document.getElementById('fase').value + ' / Semester ' + semester + ' (' + tp + ')';
       document.getElementById('resMapel').innerText = mapel;
       document.getElementById('resWaktu').innerText = waktu;
       document.getElementById('resMateri').innerText = materi;
       
       document.getElementById('resProfilList').innerHTML = profilHTML;
       document.getElementById('resPancaCintaList').innerHTML = pancaCintaHTML;

       document.getElementById('resTujuan').innerText = tujuan;
       document.getElementById('resPemantik').innerText = pemantik;
       document.getElementById('resModelName').innerText = model;
       document.getElementById('resSubMateri').innerText = materi;
       document.getElementById('resMetodeText').innerText = metode;
       document.getElementById('resLkpdTopik').innerText = materi;
       document.getElementById('signGuru').innerText = guru;

       document.getElementById('emptyState').classList.add('hidden');
       document.getElementById('resultToolbar').classList.remove('hidden');
       document.getElementById('outputContainer').classList.remove('hidden');

       if (window.innerWidth < 1024) {
           document.getElementById('outputContainer').scrollIntoView({ behavior: 'smooth' });
       }
   }

   function resetForm() {
       document.getElementById('modulForm').reset();
       document.getElementById('kelas').innerHTML = '<option value="" disabled selected>-- Jenjang Dulu --</option>';
       document.getElementById('fase').value = '';
       document.getElementById('emptyState').classList.remove('hidden');
       document.getElementById('resultToolbar').classList.add('hidden');
       document.getElementById('outputContainer').classList.add('hidden');
   }

   function copyTextResult() {
       const textContent = document.getElementById('outputContainer').innerText;
       navigator.clipboard.writeText(textContent).then(() => {
           alert('Berjaya! Seluruh teks Modul Ajar telah disalin ke clipboard.');
       }).catch(err => {
           alert('Gagal menyalin teks: ' + err);
       });
   }

   function exportWordDoc() {
       const htmlContent = document.getElementById('outputContainer').innerHTML;
       const header = "<html xmlns:o='urn:schemas-microsoft-com:office:office' xmlns:w='urn:schemas-microsoft-com:office:word' xmlns='http://www.w3.org/TR/REC-html40'><head><meta charset='utf-8'><title>Modul Ajar</title></head><body>";
       const footer = "</body></html>";
       const sourceHTML = header + htmlContent + footer;
       
       const source = 'data:application/vnd.ms-word;charset=utf-8,' + encodeURIComponent(sourceHTML);
       const fileDownload = document.createElement('a');
       document.body.appendChild(fileDownload);
       fileDownload.href = source;
       fileDownload.download = 'Modul_Ajar_Lengkap.doc';
       fileDownload.click();
       document.body.removeChild(fileDownload);
   }
